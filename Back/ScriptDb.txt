-- ============================
--        TABLE : User
-- ============================
CREATE TABLE [User] (
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Username VARCHAR(50) NOT NULL UNIQUE,
    Email VARCHAR(100) NOT NULL UNIQUE,
    PasswordHash VARCHAR(255) NOT NULL,

    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),

    IsDeleted BIT NOT NULL DEFAULT 0,
    DeletedAt DATETIME NULL
);

-- ============================
--        TABLE : Set
-- ============================
CREATE TABLE [Set] (
    Id VARCHAR(50) PRIMARY KEY,       -- ex: "sv1"
    Name VARCHAR(100) NOT NULL,
    Series VARCHAR(100),
    ReleaseDate DATE,
    LogoUrl TEXT,
    SymbolUrl TEXT
);

-- ============================
--        TABLE : Card
-- ============================
CREATE TABLE Card (
    Id VARCHAR(50) PRIMARY KEY,       -- ex: "sv1-001"
    Name VARCHAR(100) NOT NULL,
    Supertype VARCHAR(50),
    Subtype VARCHAR(50),
    Rarity VARCHAR(50),
    Hp INT,
    ImageUrl TEXT,
    SetId VARCHAR(50) NOT NULL,
    FOREIGN KEY (SetId) REFERENCES [Set](Id)
);

-- ============================
--        TABLE : Price
-- ============================
CREATE TABLE Price (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CardId VARCHAR(50) NOT NULL,
    Source VARCHAR(50) NOT NULL,      -- "cardmarket", "tcgplayer"
    Low DECIMAL(10,2),
    Mid DECIMAL(10,2),
    High DECIMAL(10,2),
    UpdatedAt DATETIME NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (CardId) REFERENCES Card(Id)
);

-- ============================
--   TABLE : CollectionCard
-- ============================
CREATE TABLE CollectionCard (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL,
    CardId VARCHAR(50) NOT NULL,
    Quantity INT NOT NULL DEFAULT 1,
    Condition VARCHAR(20),            -- Mint, NM, Played...
    AddedAt DATETIME NOT NULL DEFAULT GETDATE(),
    FOREIGN KEY (UserId) REFERENCES [User](Id),
    FOREIGN KEY (CardId) REFERENCES Card(Id)
);

-- ============================
--        TABLE : Attack
-- ============================
CREATE TABLE Attack (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CardId VARCHAR(50) NOT NULL,
    Name VARCHAR(100) NOT NULL,
    Cost VARCHAR(200),                -- JSON ou CSV
    Damage VARCHAR(20),
    Text TEXT,
    FOREIGN KEY (CardId) REFERENCES Card(Id)
);

-- ============================
--       TABLE : Weakness
-- ============================
CREATE TABLE Weakness (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CardId VARCHAR(50) NOT NULL,
    Type VARCHAR(50),
    Value VARCHAR(10),
    FOREIGN KEY (CardId) REFERENCES Card(Id)
);

-- ============================
--      TABLE : Resistance
-- ============================
CREATE TABLE Resistance (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    CardId VARCHAR(50) NOT NULL,
    Type VARCHAR(50),
    Value VARCHAR(10),
    FOREIGN KEY (CardId) REFERENCES Card(Id)
);

-- ============================
--        TABLE : Type
-- ============================
CREATE TABLE Type (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Name VARCHAR(50) UNIQUE NOT NULL
);

-- ============================
--     TABLE : CardType (N:N)
-- ============================
CREATE TABLE CardType (
    CardId VARCHAR(50) NOT NULL,
    TypeId INT NOT NULL,
    PRIMARY KEY (CardId, TypeId),
    FOREIGN KEY (CardId) REFERENCES Card(Id),
    FOREIGN KEY (TypeId) REFERENCES Type(Id)
);
