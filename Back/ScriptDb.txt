-- ============================
--        TABLE : User
-- ============================
CREATE TABLE [User] (
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Username VARCHAR(50) NOT NULL UNIQUE,
    Email VARCHAR(100) NOT NULL UNIQUE,
    PasswordHash VARCHAR(255) NOT NULL,

    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),

    IsDeleted BIT NOT NULL DEFAULT 0,
    DeletedAt DATETIME NULL
);

-- ============================
--        TABLE : Set
-- ============================
CREATE TABLE [Set] ( 
Id NVARCHAR(50) 
NOT NULL PRIMARY KEY, 
Name NVARCHAR(200) NOT NULL, 
Logo NVARCHAR(500) NULL, 
Symbol NVARCHAR(500) NULL, 
CardCountTotal INT NOT NULL, 
CardCountOfficial INT NOT NULL, 
);

-- ============================
--        TABLE : Card
-- ============================
CREATE TABLE Card
(
    Id NVARCHAR(100) NOT NULL PRIMARY KEY,

    Name NVARCHAR(250) NOT NULL,

    SetId NVARCHAR(50) NOT NULL,

    LocalId NVARCHAR(50) NULL,

    Image NVARCHAR(500) NULL,

    FOREIGN KEY (SetId)
        REFERENCES [Set](Id)
        ON DELETE CASCADE
);


-- ============================
--        TABLE : UserCard 
-- ============================
CREATE TABLE UserCard (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT NOT NULL,
    CardId NVarchar(100) NOT NULL,
    CreatedAt DATETIME NOT NULL DEFAULT GETDATE(),

    CONSTRAINT FK_UserCards_User FOREIGN KEY (UserId) REFERENCES [User](Id),
    CONSTRAINT FK_UserCards_Card FOREIGN KEY (CardId) REFERENCES Card(Id)
);
